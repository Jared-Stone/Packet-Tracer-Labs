 ##Core-R1

en
conf t
hostname Core-R1
username jstone privilege 15 secret cisco
service password-encryption
enable secret class
line vty 0 15
transport input ssh
login local 
exit
line console 0
login local 
exit
lldp run
access-list 10 permit 10.0.0.0 0.255.255.255

int g8/0
description CoreR1-CoreSW1
ip address 10.255.255.1 255.255.255.252
ip nat inside 
no shutdown
int g9/0
description CoreR1-CoreSW2
ip address 10.255.255.5 255.255.255.252
ip nat inside
no shutdown
int f0/0
description CoreR1-ISP
ip address 203.0.113.1 255.255.255.252
ip nat outside
no shutdown
exit

ip route 0.0.0.0 0.0.0.0 203.0.113.2

router ospf 1
router-id 1.1.1.1
network 10.0.0.0 0.255.255.255 area 0
default-information originate

ip dhcp pool SALES-VLAN10
network 10.10.0.0 255.255.255.0
default-router 10.10.0.1
dns-server 8.8.8.8
exit
ip dhcp pool IT-VLAN20
network 10.20.0.0 255.255.255.0
default-router 10.20.0.1
dns-server 8.8.8.8
exit
ip dhcp pool ADMIN-VLAN30
network 10.30.0.0 255.255.255.0
default-router 10.30.0.1 
dns-server 8.8.8.8
exit
ip dhcp excluded-address 10.10.0.1 10.10.0.10
ip dhcp excluded-address 10.20.0.1 10.20.0.10
ip dhcp excluded-address 10.30.0.1 10.30.0.10

ip nat inside source list 10 interface g8/0 overload
ip nat inside source list 10 interface g9/0 overload
end
write
